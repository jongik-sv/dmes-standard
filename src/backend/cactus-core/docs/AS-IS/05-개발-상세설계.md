# Cactus 개발 상세 설계

> 버전: 1.0.0
> 일시: 2026-03-23 (원본) / 2026-04-26 (현행 사실 박스 갱신)
> 참조: 01-데이터포맷-명세.md, 03-프로젝트-모듈-설계.md
>
> **현행 사실 (마이그레이션 완료, 2026-04-26)**
> - 패키지: 본 문서 본문의 `com.dongkuk.cactus.*` 코드 예시는 모두 현행 `com.dongkuk.dmes.cactus.*` 로 읽는다. (group `com.dongkuk` → `com.dongkuk.dmes`)
> - 산출물: cactus-core 는 `java-library + maven-publish` 로 발행, group `com.dongkuk.dmes`, artifactId `cactus-core`.
> - Spring Boot BOM: 4.0.6 (Spring 7.0.7). 본문의 4.0.3 표기는 옛 시점 값.
> - 멀티 모듈: 옛 `cactus/cactus-core, cactus-oasis` 구조는 폐기. 현재 cactus-core 는 단일 라이브러리 모듈이며 OASIS 통합은 `com.dongkuk.dmes.cactus.oasis.*` 패키지로 cactus-core 안에 흡수되었다.
> - AutoConfiguration imports (META-INF): `CactusAutoConfiguration`, `SecurityAutoConfiguration`, `CactusAuthAutoConfiguration`, `CactusWebSecurityAutoConfiguration`, `AuditAutoConfiguration`, `OasisAutoConfiguration`.
> - OasisController 매핑은 `/oasis/{serviceId}/{action}` 으로 고정. 옛 `/${cactus.oasis.service-group}/api` 매핑은 폐기되고 `cactus.oasis.service-group` 프로퍼티는 BPMN 라우팅/로깅 식별 용도로만 유지된다.
> - cactus-core 본체에서 AuthController 는 호출자 0 데드코드로 삭제. portal 자체 PortalAuthController(`com.dongkuk.dmes.mcm.*`) 가 `/api/auth/**` 담당.
> - 신규 표준 필터: ClientKeyFilter(`X-Backend-Client-Key`, env `BACKEND_CLIENT_KEY`), RequestIdFilter.
>
> 본 문서의 코드 예시는 **원본 설계 시점(2026-03-23) 그대로 보존**되어 있으며, 현행 패키지/그룹/매핑은 위 박스를 기준으로 읽는다.

### 3.3 TxIdGenerator.java

```java
package com.dongkuk.cactus.common;

import java.time.LocalDateTime;
import java.time.format.DateTimeFormatter;
import java.util.concurrent.ThreadLocalRandom;

/**
 * 요청 추적 ID 생성기.
 * 형식: {userId}-{menuId}-{yyyyMMddHHmmss}-{random3}
 * 예: user01-PROD001-20260323153000-a7f
 */
public class TxIdGenerator {

    private static final DateTimeFormatter FMT = DateTimeFormatter.ofPattern("yyyyMMddHHmmss");
    private static final String HEX = "0123456789abcdef";

    private TxIdGenerator() {}

    public static String generate(String userId, String menuId) {
        LocalDateTime now = LocalDateTime.now();
        String user = (userId != null && !userId.isEmpty()) ? userId : "SYSTEM";
        String menu = (menuId != null && !menuId.isEmpty()) ? menuId : "UNKNOWN";
        String suffix = randomHex(3);
        return user + "-" + menu + "-" + now.format(FMT) + "-" + suffix;
    }

    private static String randomHex(int length) {
        StringBuilder sb = new StringBuilder(length);
        for (int i = 0; i < length; i++) {
            sb.append(HEX.charAt(ThreadLocalRandom.current().nextInt(HEX.length())));
        }
        return sb.toString();
    }
}
```

### 3.4 CactusRequest.java

```java
package com.dongkuk.cactus.common;

import com.dongkuk.cactus.converter.GridConverter;
import com.fasterxml.jackson.databind.ObjectMapper;

import java.util.Collections;
import java.util.List;
import java.util.Map;

/**
 * OASIS 표준 요청 포맷.
 *
 * params와 grids는 Map 기반이며, 타입 안전한 처리가 필요한 경우
 * getParams(Class), getGrid(String, Class) 메서드로 DTO 변환이 가능하다.
 */
public class CactusRequest {

    private static final ObjectMapper MAPPER = new ObjectMapper();

    private RequestMeta meta;
    private Map<String, Object> params;
    private Map<String, GridData> grids;

    // === 기본 getter/setter ===

    public RequestMeta getMeta() { return meta; }
    public void setMeta(RequestMeta meta) { this.meta = meta; }

    public Map<String, Object> getParams() { return params; }
    public void setParams(Map<String, Object> params) { this.params = params; }

    public Map<String, GridData> getGrids() { return grids; }
    public void setGrids(Map<String, GridData> grids) { this.grids = grids; }

    // === DTO 변환 헬퍼 ===

    /**
     * params를 지정한 DTO 타입으로 변환한다.
     *
     * @param type 변환할 DTO 클래스
     * @return 변환된 DTO 인스턴스
     * @throws IllegalArgumentException 변환 실패 시
     */
    public <T> T getParams(Class<T> type) {
        if (params == null) {
            try {
                return type.getDeclaredConstructor().newInstance();
            } catch (Exception e) {
                throw new IllegalArgumentException("params DTO 변환 실패: " + type.getSimpleName(), e);
            }
        }
        return MAPPER.convertValue(params, type);
    }

    /**
     * 그리드 rows를 Map 리스트로 반환한다.
     *
     * @param gridId 그리드 식별자
     * @return 행 데이터 리스트. 그리드가 없으면 빈 리스트
     */
    public List<Map<String, Object>> getGrid(String gridId) {
        if (grids == null || !grids.containsKey(gridId)) {
            return Collections.emptyList();
        }
        return grids.get(gridId).getRows();
    }

    /**
     * 그리드 rows를 지정한 DTO 타입 리스트로 변환한다.
     *
     * @param gridId 그리드 식별자
     * @param type 변환할 DTO 클래스
     * @return 변환된 DTO 리스트. 그리드가 없으면 빈 리스트
     * @throws IllegalArgumentException 변환 실패 시
     */
    public <T> List<T> getGrid(String gridId, Class<T> type) {
        List<Map<String, Object>> rows = getGrid(gridId);
        if (rows.isEmpty()) {
            return Collections.emptyList();
        }
        return GridConverter.convert(rows, type);
    }
}
```

### 3.11 CactusResponse.java

```java
package com.dongkuk.cactus.common;

import java.util.*;

/**
 * OASIS 표준 응답 포맷.
 * Builder 패턴으로 응답을 조립한다.
 */
public class CactusResponse {

    private ResponseMeta meta;
    private Map<String, Object> data;
    private Map<String, GridResult> grids;
    private List<ErrorDetail> errors;

    private CactusResponse() {}

    // === 정적 팩토리 ===

    /**
     * 성공 응답 빌더 생성.
     * txId는 RequestMeta에서 자동 생성한다.
     */
    public static Builder success(RequestMeta reqHeader) {
        String txId = TxIdGenerator.generate(reqMeta.getUserId(), reqMeta.getMenuId());
        return new Builder(ResponseMeta.success(txId));
    }

    public static Builder success(RequestMeta reqHeader, String message) {
        String txId = TxIdGenerator.generate(reqMeta.getUserId(), reqMeta.getMenuId());
        return new Builder(ResponseMeta.success(txId, message));
    }

    /**
     * 에러 응답 빌더 생성.
     */
    public static Builder error(RequestMeta reqHeader, String code, String message) {
        String txId = TxIdGenerator.generate(reqMeta.getUserId(), reqMeta.getMenuId());
        return new Builder(ResponseMeta.error(txId, code, message));
    }

    // === Builder ===

    public static class Builder {
        private final ResponseMeta meta;
        private Map<String, Object> data;
        private Map<String, GridResult> grids;
        private List<ErrorDetail> errors;

        Builder(ResponseMeta meta) {
            this.meta = meta;
        }

        public Builder datum(String key, Object value) {
            if (this.data == null) this.data = new LinkedHashMap<>();
            this.data.put(key, value);
            return this;
        }

        public Builder data(Map<String, Object> data) {
            this.data = data;
            return this;
        }

        public Builder grid(String gridId, List<Map<String, Object>> rows) {
            if (this.grids == null) this.grids = new LinkedHashMap<>();
            this.grids.put(gridId, new GridResult(rows));
            return this;
        }

        public Builder grid(String gridId, List<ColumnMeta> columns, List<Map<String, Object>> rows) {
            if (this.grids == null) this.grids = new LinkedHashMap<>();
            this.grids.put(gridId, new GridResult(columns, rows));
            return this;
        }

        public Builder error(ErrorDetail error) {
            if (this.errors == null) this.errors = new ArrayList<>();
            this.errors.add(error);
            return this;
        }

        public Builder errors(List<ErrorDetail> errors) {
            this.errors = errors;
            return this;
        }

        public CactusResponse build() {
            CactusResponse r = new CactusResponse();
            r.meta = this.meta;
            r.data = this.data;
            r.grids = this.grids;
            r.errors = this.errors;
            return r;
        }
    }

    // === getter ===

    public ResponseMeta getMeta() { return meta; }
    public Map<String, Object> getData() { return data; }
    public Map<String, GridResult> getGrids() { return grids; }
    public List<ErrorDetail> getErrors() { return errors; }
}
```

```java
package com.dongkuk.cactus.converter;

import com.fasterxml.jackson.databind.DeserializationFeature;
import com.fasterxml.jackson.databind.ObjectMapper;

import java.util.List;
import java.util.Map;
import java.util.stream.Collectors;

/**
 * Map<String, Object> ↔ DTO 변환 유틸.
 * Jackson ObjectMapper를 사용하여 변환한다.
 */
public class GridConverter {

    private static final ObjectMapper MAPPER = new ObjectMapper()
            .configure(DeserializationFeature.FAIL_ON_UNKNOWN_PROPERTIES, false);

    private GridConverter() {}

    /**
     * Map 리스트를 DTO 리스트로 변환한다.
     * 매핑되지 않는 필드는 무시한다.
     *
     * @param rows Map 리스트
     * @param type 변환할 DTO 클래스
     * @return 변환된 DTO 리스트
     * @throws IllegalArgumentException 변환 실패 시
     */
    public static <T> List<T> convert(List<Map<String, Object>> rows, Class<T> type) {
        try {
            return rows.stream()
                    .map(GridConverter::normalizeEmptyStrings)
                    .map(row -> MAPPER.convertValue(row, type))
                    .collect(Collectors.toList());
        } catch (IllegalArgumentException e) {
            throw new IllegalArgumentException(
                    "그리드 행 → " + type.getSimpleName() + " 변환 실패: " + e.getMessage(), e);
        }
    }

    /**
     * 빈문자열("")을 null로 정규화한다.
     * 숫자/날짜/불린 필드에 ""이 들어오면 DTO 변환 시 예외가 발생하므로,
     * 변환 전에 "" → null로 치환한다.
     */
    private static Map<String, Object> normalizeEmptyStrings(Map<String, Object> row) {
        Map<String, Object> normalized = new HashMap<>(row);
        normalized.replaceAll((key, value) -> {
            if ("".equals(value)) return null;
            return value;
        });
        return normalized;
    }

    /**
     * DTO 리스트를 Map 리스트로 변환한다.
     * CactusResponse에 그리드를 담을 때 사용한다.
     *
     * @param dtoList DTO 리스트
     * @return Map 리스트
     */
    @SuppressWarnings("unchecked")
    public static <T> List<Map<String, Object>> toMapList(List<T> dtoList) {
        return dtoList.stream()
                .map(dto -> MAPPER.convertValue(dto, Map.class))
                .map(map -> (Map<String, Object>) map)
                .collect(Collectors.toList());
    }
}
```

```java
package com.dongkuk.cactus.common;

import org.slf4j.Logger;
import org.slf4j.LoggerFactory;
import org.springframework.http.ResponseEntity;
import org.springframework.web.bind.annotation.ExceptionHandler;
import org.springframework.web.bind.annotation.RestControllerAdvice;

/**
 * 전역 예외 핸들러.
 * BusinessException → CactusResponse 에러 응답으로 변환한다.
 * MDC에 저장된 txId를 응답에 포함한다.
 */
@RestControllerAdvice
public class GlobalExceptionHandler {

    private static final Logger log = LoggerFactory.getLogger(GlobalExceptionHandler.class);

    @ExceptionHandler(BusinessException.class)
    public ResponseEntity<CactusResponse> handleBusinessException(BusinessException e) {
        log.warn("BusinessException: [{}] {}", e.getErrorCode().getCode(), e.getMessage());

        String txId = MDC.get("txId");
        ResponseMeta meta = ResponseMeta.error(
                txId, e.getErrorCode().getCode(), e.getMessage());

        CactusResponse response = new CactusResponse.Builder(meta)
                .errors(e.getErrors())
                .build();

        return ResponseEntity
                .status(e.getErrorCode().getHttpStatus())
                .body(response);
    }

    @ExceptionHandler(Exception.class)
    public ResponseEntity<CactusResponse> handleException(Exception e) {
        log.error("Unhandled exception", e);

        String txId = MDC.get("txId");
        ResponseMeta meta = ResponseMeta.error(
                txId,
                ErrorCode.UNKNOWN_ERROR.getCode(),
                ErrorCode.UNKNOWN_ERROR.getDefaultMessage());

        CactusResponse response = new CactusResponse.Builder(meta).build();

        return ResponseEntity
                .status(500)
                .body(response);
    }
}
```

### 3.16 TxIdFilter.java

```java
package com.dongkuk.cactus.filter;

import jakarta.servlet.FilterChain;
import jakarta.servlet.ServletException;
import jakarta.servlet.http.HttpServletRequest;
import jakarta.servlet.http.HttpServletResponse;
import org.slf4j.MDC;
import org.springframework.core.Ordered;
import org.springframework.core.annotation.Order;
import org.springframework.stereotype.Component;
import org.springframework.web.filter.OncePerRequestFilter;

import java.io.IOException;
import java.util.UUID;

/**
 * 요청 진입 시 MDC에 txId를 설정하는 필터.
 *
 * 동작 흐름:
 * 1. Filter 진입 → 임시 UUID를 MDC에 설정 (body 파싱 전이라 userId/menuId를 모름)
 * 2. Controller 진입 → CactusRequest 파싱 후 정식 txId({userId}-{menuId}-{yyyyMMddHHmmss}-{random3})로 교체
 * 3. 응답 완료 → MDC 클리어
 *
 * logback 설정:
 *   %d{yyyy-MM-dd HH:mm:ss.SSS} [%thread] [%X{txId}] %-5level %logger{36} - %msg%n
 */
@Component
@Order(Ordered.HIGHEST_PRECEDENCE)
public class TxIdFilter extends OncePerRequestFilter {

    @Override
    protected void doFilterInternal(HttpServletRequest request,
                                     HttpServletResponse response,
                                     FilterChain chain) throws ServletException, IOException {
        try {
            // 1. 임시 UUID로 MDC 설정 (Filter 구간 로그용)
            String tempTxId = UUID.randomUUID().toString().substring(0, 8);
            MDC.put("txId", tempTxId);

            // 2. 요청 처리 (Controller에서 정식 txId로 교체됨)
            chain.doFilter(request, response);

        } finally {
            // 3. 요청 완료 후 MDC 클리어
            MDC.clear();
        }
    }
}
```

### 4.4 CactusRequestConverter.java

```java
package com.dongkuk.cactus.oasis;

import com.dongkuk.cactus.common.CactusRequest;
import com.dongkuk.cactus.common.RequestMeta;
import com.dongkuk.oasis.TypedObject;

import java.util.HashMap;
import java.util.Map;

/**
 * CactusRequest → Map<String, TypedObject> 변환기.
 */
public class CactusRequestConverter {

    /**
     * CactusRequest를 oasis ServiceContext 입력 형태로 변환한다.
     *
     * 변환 규칙:
     * - action → TypedObject
     * - meta 필드 (userId, menuId) → 각각 TypedObject
     * - params의 각 key-value → TypedObject
     * - grids의 각 gridId → TypedObject(List<Map>)
     */
    public Map<String, TypedObject> convert(CactusRequest request, String action) {
        Map<String, TypedObject> map = new HashMap<>();

        // action (BPMN 게이트웨이 분기용)
        map.put("action", new TypedObject(action));

        // meta
        RequestMeta meta = request.getMeta();
        if (meta != null) {
            map.put("userId", new TypedObject(meta.getUserId()));
            map.put("menuId", new TypedObject(meta.getMenuId()));
        }

        // params
        if (request.getParams() != null) {
            request.getParams().forEach((key, value) ->
                    map.put(key, new TypedObject(value)));
        }

        // grids
        if (request.getGrids() != null) {
            request.getGrids().forEach((key, gridData) ->
                    map.put(key, new TypedObject(gridData.getRows())));
        }

        return map;
    }
}
```

### 4.5 CactusResponseConverter.java

```java
package com.dongkuk.cactus.oasis;

import com.dongkuk.cactus.common.*;
import com.dongkuk.oasis.TypedObject;
import com.dongkuk.oasis.service.ServiceResult;
import com.dongkuk.oasis.service.ServiceResultCode;

import java.util.HashMap;
import java.util.List;
import java.util.Map;

/**
 * ServiceResult → CactusResponse 변환기.
 */
public class CactusResponseConverter {

    /**
     * ServiceResult를 CactusResponse로 변환한다.
     *
     * 변환 규칙:
     * - SUCCESS → results 중 List 타입은 grids, 단일 값은 data에 매핑
     * - USER_ERROR → meta.success=false, code="E001"
     * - SYSTEM_ERROR → meta.success=false, code="S001"
     */
    public CactusResponse convert(ServiceResult result, String txId) {
        if (result.serviceResultCode() == ServiceResultCode.SUCCESS) {
            return convertSuccess(result, txId);
        } else {
            return convertError(result, txId);
        }
    }

    @SuppressWarnings("unchecked")
    private CactusResponse convertSuccess(ServiceResult result, String txId) {
        Map<String, Object> data = new HashMap<>();
        Map<String, GridResult> grids = new HashMap<>();

        if (result.results() != null) {
            result.results().forEach((key, typedObject) -> {
                Object value = typedObject.getObject();
                if (value instanceof List) {
                    grids.put(key, new GridResult((List<Map<String, Object>>) value));
                } else {
                    data.put(key, value);
                }
            });
        }

        CactusResponse.Builder builder = new CactusResponse.Builder(ResponseMeta.success(txId));
        if (!data.isEmpty()) builder.data(data);
        return builder.build();
    }

    private CactusResponse convertError(ServiceResult result, String txId) {
        String code = (result.serviceResultCode() == ServiceResultCode.USER_ERROR) ? "E001" : "S001";
        String message = result.serviceResultMessage() != null
                ? result.serviceResultMessage()
                : "오류가 발생했습니다.";

        return new CactusResponse.Builder(ResponseMeta.error(txId, code, message)).build();
    }
}
```

### 4.6 OasisServiceExecutor.java

```java
package com.dongkuk.cactus.oasis;

import com.dongkuk.cactus.common.*;
import com.dongkuk.oasis.TypedObject;
import com.dongkuk.oasis.context.DefaultServiceContext;
import com.dongkuk.oasis.context.ServiceContext;
import com.dongkuk.oasis.context.SpringApplicationContext;
import com.dongkuk.oasis.service.ServiceResult;
import com.dongkuk.oasis.service.ServiceStarter;
import org.slf4j.Logger;
import org.slf4j.LoggerFactory;
import org.slf4j.MDC;
import org.springframework.context.ApplicationContext;

import java.util.Map;

/**
 * CactusRequest → oasis 서비스 실행 → CactusResponse 전체 흐름을 담당한다.
 * txId 생성 후 MDC에 교체하여 이후 모든 로그에 정식 txId가 출력되도록 한다.
 */
public class OasisServiceExecutor {

    private static final Logger log = LoggerFactory.getLogger(OasisServiceExecutor.class);

    private final ServiceStarter serviceStarter;
    private final ApplicationContext springApplicationContext;
    private final CactusRequestConverter requestConverter;
    private final CactusResponseConverter responseConverter;

    public OasisServiceExecutor(
            ServiceStarter serviceStarter,
            ApplicationContext springApplicationContext,
            CactusRequestConverter requestConverter,
            CactusResponseConverter responseConverter) {
        this.serviceStarter = serviceStarter;
        this.springApplicationContext = springApplicationContext;
        this.requestConverter = requestConverter;
        this.responseConverter = responseConverter;
    }

    /**
     * 서비스를 실행하고 CactusResponse를 반환한다.
     *
     * @param serviceGroup 서비스 그룹 (로깅용)
     * @param serviceId BPMN 서비스 ID
     * @param action 액션 (search, save, delete 등)
     * @param request CactusRequest
     * @return CactusResponse
     */
    public CactusResponse execute(String serviceGroup, String serviceId, String action, CactusRequest request) {
        RequestMeta reqMeta = request.getMeta();
        String txId = TxIdGenerator.generate(reqMeta.getUserId(), reqMeta.getMenuId());

        // MDC의 임시 UUID를 정식 txId로 교체
        MDC.put("txId", txId);

        log.info("{}/{}/{}", serviceGroup, serviceId, action);

        try {
            // 1. CactusRequest → TypedObject Map
            Map<String, TypedObject> inputs = requestConverter.convert(request, action);

            // 2. ServiceContext 생성
            com.dongkuk.oasis.context.ApplicationContext oasisAppCtx =
                    new SpringApplicationContext(springApplicationContext);
            ServiceContext sc = new DefaultServiceContext(oasisAppCtx, inputs);

            // 3. 서비스 실행
            ServiceResult result = serviceStarter.start(serviceId, sc);

            // 4. ServiceResult → CactusResponse
            return responseConverter.convert(result, txId);

        } catch (BusinessException e) {
            log.warn("[{}] BusinessException: {}", txId, e.getMessage());
            return new CactusResponse.Builder(
                    ResponseMeta.error(txId, e.getErrorCode().getCode(), e.getMessage()))
                    .errors(e.getErrors())
                    .build();

        } catch (Exception e) {
            log.error("[{}] Unexpected error", txId, e);
            return new CactusResponse.Builder(
                    ResponseMeta.error(txId, ErrorCode.UNKNOWN_ERROR.getCode(), e.getMessage()))
                    .build();
        }
    }
}
```
